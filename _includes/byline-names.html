{%- comment -%} Formats one byline name list for the homepage (see byline-home.html). Params: include.raw - raw " and "-joined name string (a front-matter field) include.prefix_with - prefix used when "Rodighiero" IS in the list (e.g. "with ") include.prefix_by - prefix used when "Rodighiero" is NOT in the list (e.g. "edited by ") Strips "Rodighiero", joins the rest with ", " + a final " and " (no Oxford comma). Outputs nothing when raw is empty or Dario is the only name. {%- endcomment -%} {%- assign me = "Rodighiero" -%} {%- comment -%} Liquid has no array literal, so names are appended with a sentinel and split back apart. "~~~" is the convention across the byline includes (see also byline-home.html, publication-cite.html); it never occurs in the front matter it splits. {%- endcomment -%} {%- assign SEP = "~~~" -%} {%- if include.raw -%} {%- assign raw_arr = include.raw | split: " and " -%} {%- assign had_me = false -%} {%- assign kept = "" -%} {%- for n in raw_arr -%} {%- assign n_trim = n | strip -%} {%- if n_trim contains me -%} {%- assign had_me = true -%} {%- else -%} {%- if kept == "" -%}{%- assign kept = n_trim -%} {%- else -%}{%- assign kept = kept | append: SEP | append: n_trim -%}{%- endif -%} {%- endif -%} {%- endfor -%} {%- if kept != "" -%} {%- assign arr = kept | split: SEP -%} {%- capture list -%}{%- include byline-join.html items=arr -%}{%- endcapture -%} {%- if had_me -%}{{ include.prefix_with }}{{ list }} {%- else -%}{{ include.prefix_by }}{{ list }}{%- endif -%} {%- endif -%} {%- endif -%}
